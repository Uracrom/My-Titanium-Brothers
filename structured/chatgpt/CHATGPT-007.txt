```python
    """Shift ASCII shape horizontally by offset within given width."""
    shifted = []
    for row in shape:
        padding_left = " " * offset
        padding_right = " " * (width - offset - len(row))
        shifted.append(padding_left + row + padding_right)
    return shifted

        time.sleep(1 / fps)
```

```python
# server.py
import asyncio
import json
import websockets
import time

# Simple frames (you can replace with larger grids or dynamic generator)
frames = [
    ["........*...........", ".......***..........", "......*****.........", ".......***..........", "........*..........."],
    [".........*..........", "........***.........", ".......*****........", "........***.........", ".........*.........."],
    ["..........*.........", ".........***........", "........*****.......", ".........***........", "..........*........."],
    ["...........*........", "..........***.......", ".........*****......", "..........***.......", "...........*........"],
    ["............*.......", "...........***......", "..........*****.....", "...........***......", "............*......."]
]

# Bounce path (forward then backward)
bounce = frames + frames[-2:0:-1]

FPS = 30
FRAME_DELAY = 1.0 / FPS
PORT = 6789

connected = set()

async def producer(ws):
    """Continuously produce frames to a connected websocket."""
    while True:
        for frame in bounce:
            payload = json.dumps({"frame": frame})
            try:
                await ws.send(payload)
            except websockets.ConnectionClosed:
                return
            await asyncio.sleep(FRAME_DELAY)

async def handler(ws, path):
    print(f"Client connected: {ws.remote_address}")
    connected.add(ws)
    try:
        await producer(ws)
    except asyncio.CancelledError:
        pass
    finally:
        connected.remove(ws)
        print(f"Client disconnected: {ws.remote_address}")

async def main():
    print(f"Starting ASCII streaming server on ws://localhost:{PORT}")
    async with websockets.serve(handler, "0.0.0.0", PORT):
        await asyncio.Future()  # run forever

if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print("Server stopped.")
```
