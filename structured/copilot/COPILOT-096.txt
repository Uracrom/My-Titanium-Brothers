```javascript
// Starfield animation
const canvas = document.getElementById('starfield');
const ctx = canvas.getContext('2d');
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

let stars = [];
for (let i = 0; i < 300; i++) {
  stars.push({
    x: Math.random() * canvas.width,
    y: Math.random() * canvas.height,
    radius: Math.random() * 1.5,
    speed: Math.random() * 0.2 + 0.05
  });
}

function animateStars() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = 'white';
  stars.forEach(star => {
    ctx.beginPath();
    ctx.arc(star.x, star.y, star.radius, 0, Math.PI * 2);
    ctx.fill();
    star.y += star.speed;
    if (star.y > canvas.height) star.y = 0;
  });
  requestAnimationFrame(animateStars);
}
animateStars();

// Comment interaction
const commentBtn = document.getElementById('commentBtn');
const commentModal = document.getElementById('commentModal');
const aboutBox = document.getElementById('aboutBox');

commentBtn.addEventListener('click', () => {
  document.body.style.cursor = 'crosshair';
  commentModal.classList.remove('hidden');

  document.querySelectorAll('.dialogue p').forEach(p => {
    p.addEventListener('mouseup', () => {
      const selection = window.getSelection().toString();
      if (selection) {
        aboutBox.value = selection;
        document.body.style.cursor = 'default';
      }
    });
  });
});

const express = require('express');
const sqlite3 = require('sqlite3').verbose();
const bodyParser = require('body-parser');
const app = express();
const db = new sqlite3.Database('./database.db');

app.use(bodyParser.json());
app.use(express.static('public'));

// Create tables
db.run(`CREATE TABLE IF NOT EXISTS comments (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  thread TEXT,
  about TEXT,
  comment TEXT,
  alias TEXT,
  timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
)`);

// Get comments for a thread
app.get('/comments/:thread', (req, res) => {
  db.all(`SELECT * FROM comments WHERE thread = ?`, [req.params.thread], (err, rows) => {
    res.json(rows);
  });
});

// Post a comment
app.post('/comment', (req, res) => {
  const { thread, about, comment, alias } = req.body;
  db.run(`INSERT INTO comments (thread, about, comment, alias) VALUES (?, ?, ?, ?)`,
    [thread, about, comment, alias], function(err) {
      res.json({ success: true, id: this.lastID });
    });
});

app.listen(3000, () => console.log('Server running on http://localhost:3000'));
```
