```python
import gradio as gr
import cv2
import base64
import requests
import os

Set your xAI API key
API_KEY = os.getenv("XAI_API_KEY")  # Export it in your environment

def analyze_frame(image, prompt):
Encode image to base64
    _, img_encoded = cv2.imencode('.jpg', image)
    img_base64 = base64.b64encode(img_encoded).decode('utf-8')

Prepare payload for Grok API (adjust endpoint/model as per docs)
    payload = {
        "model": "grok-4",  # Or latest vision-capable model
        "messages": [
            {"role": "user", "content": [
                {"type": "text", "text": prompt},
                {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{img_base64}"}}
            ]}
        ],
        "max_tokens": 300
    }

    headers = {"Authorization": f"Bearer {API_KEY}"}
    response = requests.post("https://api.x.ai/v1/chat/completions", headers=headers, json=payload)
    return response.json()["choices"][0]["message"]["content"]

Gradio interface
with gr.Blocks() as demo:
    gr.Markdown("Webcam to Grok Vision")
    with gr.Row():
        webcam = gr.Image(source="webcam", streaming=True, label="Live Webcam")
        output = gr.Textbox(label="Grok's Response")
    prompt = gr.Textbox(label="Prompt (e.g., 'Describe this scene')")
    submit = gr.Button("Analyze")

    submit.click(analyze_frame, inputs=[webcam, prompt], outputs=output)

demo.launch()

Artifact: Enhanced Webcam-to-LLM Script
If the details above are overwhelming, here's a self-contained "artifact"—an updated Python script inspired by the research (e.g., adding object detection from YOLO and multimodal prompting). ```python
import cv2
import base64
import requests
import os
from ultralytics import YOLO  # Add YOLO for object detection (pip install ultralytics)

Set your xAI API key
API_KEY = os.getenv("XAI_API_KEY")

Load YOLO model for pre-processing
model = YOLO("yolov8n.pt")  # Nano model for speed

def analyze_frame(image, prompt):
Run YOLO detection
    results = model(image)
    detections = results[0].boxes.data.tolist()  # Extract bounding boxes/objects

Encode image
    _, img_encoded = cv2.imencode('.jpg', image)
    img_base64 = base64.b64encode(img_encoded).decode('utf-8')

Enhanced prompt with detections
    enhanced_prompt = f"{prompt}\nDetected objects: {detections}"

    payload = {
        "model": "grok-4",
        "messages": [
            {"role": "user", "content": [
                {"type": "text", "text": enhanced_prompt},
                {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{img_base64}"}}
            ]}
        ],
        "max_tokens": 500
    }

    headers = {"Authorization": f"Bearer {API_KEY}"}
    response = requests.post("https://api.x.ai/v1/chat/completions", headers=headers, json=payload)
    return response.json()["choices"][0]["message"]["content"]

Simple webcam loop
cap = cv2.VideoCapture(0)
while True:
    ret, frame = cap.read()
    if not ret:
        break
    cv2.imshow('Webcam', frame)

Analyze on key press (e.g., spacebar)
    if cv2.waitKey(1) & 0xFF == ord(' '):
        result = analyze_frame(frame, "Describe this scene and suggest actions based on what's visible.")
        print(result)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```
