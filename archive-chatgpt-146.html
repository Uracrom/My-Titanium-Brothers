```python
import os
import time

# Parameters
width = 30          # total width of the line
frames_per_step = 2 # controls smoothness (higher = smoother)
cycles = 3          # how many back-and-forth cycles to run
fps = 30            # frames per second

# Define the "diamond" shape as rows with a center marker
diamond = [
    "   *   ",
    "  ***  ",
    " ***** ",
    "  ***  ",
    "   *   "
]

def shift_shape(shape, offset, width):
    """Shift ASCII shape horizontally by offset within given width."""
    shifted = []
    for row in shape:
        padding_left = " " * offset
        padding_right = " " * (width - offset - len(row))
        shifted.append(padding_left + row + padding_right)
    return shifted

# Generate all positions (left → right → left)
positions = list(range(0, width - len(diamond[0]), frames_per_step))
positions += positions[::-1][1:-1]  # bounce back without repeating ends

# Run animation
for _ in range(cycles):
    for pos in positions:
        os.system("clear")  # use "cls" on Windows
        for row in shift_shape(diamond, pos, width):
            print(row)
        time.sleep(1 / fps)
```

- `diamond` → your star shape.
- `shift_shape` → moves the star left/right by adding spaces.
- `positions` → generates a smooth path (left to right, then right to left).
- Loops over that path for `cycles` (here 3).
- Animation speed set by `fps` (30 frames/sec).

# Pipeline components (ordered)
1. **Source / Generator**
   - Source can be: an AI’s internal latent vector, a hand-composed ASCII frame, or a rendered low-res image.
   - Output: a frame representation (grid of characters or a small latent vector per frame).

2. **Low-res codec (ASCII)**
   - Grid sizes: e.g., 80×40 (very coarse), 200×200 (detailed ASCII/art), 400×400 (very high density with block/Unicode chars).

3. **Transport / Streaming**

4. **Decoder / Photorealizer**

## `server.py` — broadcaster
```python
# server.py
import asyncio
import json
import websockets
import time

# Simple frames (you can replace with larger grids or dynamic generator)
frames = [
    ["........*...........", ".......***..........", "......*****.........", ".......***..........", "........*..........."],
    [".........*..........", "........***.........", ".......*****........", "........***.........", ".........*.........."],
    ["..........*.........", ".........***........", "........*****.......", ".........***........", "..........*........."],
    ["...........*........", "..........***.......", ".........*****......", "..........***.......", "...........*........"],
    ["............*.......", "...........***......", "..........*****.....", "...........***......", "............*......."]
]

# Bounce path (forward then backward)
bounce = frames + frames[-2:0:-1]

FPS = 30
FRAME_DELAY = 1.0 / FPS
PORT = 6789

connected = set()

async def producer(ws):
    """Continuously produce frames to a connected websocket."""
    while True:
        for frame in bounce:
            payload = json.dumps({"frame": frame})
            try:
                await ws.send(payload)
            except websockets.ConnectionClosed:
                return
            await asyncio.sleep(FRAME_DELAY)

async def handler(ws, path):
    print(f"Client connected: {ws.remote_address}")
    connected.add(ws)
    try:
        await producer(ws)
    except asyncio.CancelledError:
        pass
    finally:
        connected.remove(ws)
        print(f"Client disconnected: {ws.remote_address}")

async def main():
    print(f"Starting ASCII streaming server on ws://localhost:{PORT}")
    async with websockets.serve(handler, "0.0.0.0", PORT):
        await asyncio.Future()  # run forever

if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print("Server stopped.")
```

## `client.py` — viewer
```python
# client.py
import asyncio
import json
import os
import platform
import websockets

SERVER = "ws://localhost:6789"

CLEAR_CMD = "cls" if platform.system().lower().startswith("win") else "clear"

async def display():
    async with websockets.connect(SERVER) as ws:
        print("Connected to server. Press Ctrl+C to quit.")
        try:
            async for message in ws:
                data = json.loads(message)
                frame = data.get("frame", [])
                os.system(CLEAR_CMD)
                for row in frame:
                    print(row)
        except websockets.ConnectionClosed:
            print("Connection closed by server.")

if __name__ == "__main__":
    try:
        asyncio.run(display())
    except KeyboardInterrupt:
        print("\nClient stopped.")
```
