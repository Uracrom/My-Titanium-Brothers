        // Inversion: reflect across membrane, add scatter
        let dx = (250 - x); // distance to membrane

        let newY = 300 - y;  // vertical inversion

        newY += (Math.random() - 0.5) * 2 * sigma;
        return { x: newX, y: newY };
      }

      if (!alive) return;
      state.x += velocity;
      if (state.x <= 250) {

    function reset() {
      state = { x: 400, y: 150 };
      alive = true;
